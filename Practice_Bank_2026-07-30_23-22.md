### Question 2:
1. Implement a queue data structure using a linked list in Python and demonstrate how to use it to solve the "Print Prime Numbers" problem: write a function that prints all prime numbers up to a given integer N.

**Solution:**
```
class Node:
    def __init__(self, value):
        self.value = value
        self.next = None

class Queue:
    def __init__(self):
        self.front = None
        self.rear = None

    def is_empty(self):
        return self.front == None

    def enqueue(self, value):
        new_node = Node(value)
        if self.is_empty():
            self.front = new_node
            self.rear = new_node
        else:
            self.rear.next = new_node
            self.rear = new_node

    def dequeue(self):
        if not self.is_empty():
            value = self.front.value
            self.front = self.front.next
            if self.front == None:
                self.rear = None
            return value
        else:
            return None

def print_primes(n):
    queue = Queue()
    for i in range(2, n + 1):
        queue.enqueue(i)
        while not queue.is_empty():
            num = queue.dequeue()
            if num > 1:
                for j in range(2, int(num ** 0.5) + 1):
                    if num % j == 0:
                        break
                else:
                    print(num)

print_primes(30)
```

### Question 3:
2. Create a program that uses recursion to traverse a binary tree of arbitrary depth, printing the values at each node in pre-order (root -> left -> right).

**Solution:**
```
void printPreOrder(Node* root) {
    if(root != NULL) {
        cout << root->value << " ";
        printPreOrder(root->left);
        printPreOrder(root->right);
    }
}

class Node {
public:
    int value;
    Node* left;
    Node* right;

    Node(int val) {
        value = val;
        left = NULL;
        right = NULL;
    }
};
```

### Question 4:
3. Write a Python script that simulates a game of "Tic Tac Toe" using a 3x3 grid and allows two players to take turns placing X's and O's on the board, checking for a win or draw after each move.

**Solution:**
```
board = [[' ' for _ in range(3)] for _ in range(3)]
current_player = 'X'
while True:
    print('\n'.join([' '.join(row) for row in board]))
    if current_player == 'X':
        move = input('Player X, enter your move (1-9): ')
        if board[(int(move)-1)//3][(int(move)-1)%3] == ' ':
            board[(int(move)-1)//3][(int(move)-1)%3] = 'X'
        else:
            print("Invalid move, try again.")
            continue
    else:
        move = input('Player O, enter your move (1-9): ')
        if board[(int(move)-1)//3][(int(move)-1)%3] == ' ':
            board[(int(move)-1)//3][(int(move)-1)%3] = 'O'
        else:
            print("Invalid move, try again.")
            continue
    for i in range(3):
        if board[i][0] == board[i][1] == board[i][2] != ' ':
            print('\n'.join([' '.join(row) for row in board]))
            print(f"Player {current_player} wins!")
            exit()
    for i in range(3):
        if board[0][i] == board[1][i] == board[2][i] != ' ':
            print('\n'.join([' '.join(row) for row in board]))
            print(f"Player {current_player} wins!")
            exit()
    if board[0][0] == board[1][1] == board[2][2] != ' ':
        print('\n'.join([' '.join(row) for row in board]))
        print(f"Player {current_player} wins!")
        exit()
    if board[0][2] == board[1][1] == board[2][0] != ' ':
        print('\n'.join([' '.join(row) for row in board]))
        print(f"Player {current_player} wins!")
        exit()
    if all(cell != ' ' for row in board for cell in row):
        print('\n'.join([' '.join(row) for row in board]))
        print("It's a draw!")
```
